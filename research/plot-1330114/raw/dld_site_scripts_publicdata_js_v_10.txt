var sortByProperty = function (property) {
    return function (x, y) {
        return ((x[property] === y[property]) ? 0 : ((x[property] > y[property]) ? 1 : -1));
    };
};


var getPropertyName = (property) => property + (lang == 'ar' ? "_AR" : "_EN");

var getRequestBody = (form, parameters) => {
    var formData = new FormData(form);
    var filter = {};
    formData.forEach(function (value, key) {
        if (parameters && parameters.length && parameters.findIndex(i => i == key) < 0) return;
        if ((key == 'P_FROM_DATE' || key == 'P_TO_DATE') && value) value = moment(value, 'DD/MM/YYYY').locale('en').format("MM/DD/YYYY");
        filter[key] = value;
    });
    return filter;
}

var checkValidity = (form) => {
    var r = form.checkValidity() 
    if (r === false) {
        var invalids = form.querySelectorAll(':invalid');
        for (var i of invalids) {
            if (i.validity.customError) {
                var msgElement = 'data-' + i.validationMessage + '-msg'
                $(i).siblings('.invalid-feedback').find('span').hide();
                $(i).siblings('.invalid-feedback').find('span[' + msgElement + ']').show();
            } else {
                $(i).siblings('.invalid-feedback').find('span').hide();
                $(i).siblings('.invalid-feedback').find('span[data-default]').show();
            }
        }
    }
    if (!form.classList.contains('was-validated')) {
        form.classList.add('was-validated')
    }
    return r;
}

var instance = null;
function exportToExcel() {
    $("#loading").show();
    if (!instance || !instance.getExportFilter) return;
    var details = instance.getExportFilter();

    openDataApi.exportToExcel(details.command, details, () => {
        $("#loading").hide();
    });    
}

function exportToCsv() {
    if (!instance || !instance.getExportFilter) return;
    if (checkValidity(instance.form) === false) return;
    $("#loading").show();    
    var details = instance.getExportFilter();
    openDataApi.exportToCsv(details.command, details, () => {
        $("#loading").hide();
    });  
}
function exportToPdf() {
    $("#loading").show();
    if (!instance || !instance.getExportFilter) return;
    var details = instance.getExportFilter();

    openDataApi.exportToPdf(details.command, details, () => {
        $("#loading").hide();
    });  
}

function initSection(section) {    
    if (instance) instance.destroy();
    instance = section;
    if (instance.form) instance.form.classList.remove('was-validated');
    instance.init();
}
