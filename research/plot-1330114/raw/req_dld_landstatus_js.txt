const formatNumber = function (name) {
    if (!name) return name
    try {
        return name.toString().replace(/\B(?=(\d{3})+(?!\d))/g, ",")
    } catch {
    }

    return name.toLocaleString();
}

Handlebars.registerHelper('getLocaleName', function (name) {
    return name ? LocaleUtil.getLocalValue(name) : "";
});

Handlebars.registerHelper('formatNumber', formatNumber);

Handlebars.registerHelper('formatDate', function (name) {
    return moment(name).format("DD/MM/YYYY");
});

Handlebars.registerHelper('json', function (data) {
    return JSON.stringify(data);
});

Handlebars.registerHelper('if-null-undefined', function (data, options) {
    if (data === undefined || data === null) {
        return options.fn(this);
    } else {
        return options.inverse(this);
    }
})

Handlebars.registerHelper('if-null-empty', function (data, options) {
    if (data === undefined || data === null || (data.trim && data.trim() === "")) {
        return options.fn(this);
    } else {
        return options.inverse(this);
    }
})

function getCookie(cname) {
    let name = cname + "=";
    let decodedCookie = decodeURIComponent(document.cookie);
    let ca = decodedCookie.split(';');
    for (let i = 0; i < ca.length; i++) {
        let c = ca[i];
        while (c.charAt(0) == ' ') {
            c = c.substring(1);
        }
        if (c.indexOf(name) == 0) {
            return c.substring(name.length, c.length);
        }
    }
    return '';
}

function focusLogic() {
    $('input[required],select[required],textarea[required]').on('focus', function () {
        $(this).removeClass('empty-validation').removeClass('valid-validation').addClass('focus-validation');
    }).on('focusout', function () {
        if ($(this).attr('id') == 'emiratesIdField') {
            if ($(this).val().length == 15) {
                $(this).removeClass('focus-validation').addClass('empty-validation').addClass('valid-validation');

            } else {
                $(this).removeClass('focus-validation').removeClass('valid-validation').addClass('empty-validation');
            }
        } else {
            if ($(this).val() !== '' && $(this).valid()) {
                $(this).removeClass('focus-validation').addClass('empty-validation').addClass('valid-validation');
            } else {
                $(this).removeClass('focus-validation').removeClass('valid-validation').addClass('empty-validation');
            }
        }
    })
}

function loadCertificateYears() {
    var n = (new Date()).getFullYear() - 1970 + 1;
    Array(n).fill(1).map(function (i, e) { return e + 1970 }).sort(function (a, b) { return b - a }).forEach(function (item) {
        $("#CertificateYear").append($('<option/>', {
            value: item,
            text: item
        })).trigger("chosen:updated");
    })
}

function getPropDetails(propId) {
    propInfo = {};
    var prMap = GetPropMap(propId).then(function (map) {
        $("#map-photo-modal-body").empty();
        if (map != '') {
            var $img = $("<img>", {
                src: map,
                alt: "Logo of Property status map",
                id: "map-photo-modal-myModal",
                class: "card-img"
            });
            $("#map-photo-modal-body").append($img);
            propInfo.imgSrc = map;
        }
    });
    var prXtra = GetPropExtraInfo(propId).then(function (xtra) {
        if (xtra) {
            propInfo.lat = xtra.LAT;
            propInfo.lng = xtra.LNG;
            propInfo.parcelId = xtra.PARCEL_ID;
            propInfo.propTypeId = xtra.PROPERTY_TYPE_ID;
            propInfo.propSubTypeId = xtra.PROPERTY_SUB_TYPE_ID;
            propInfo.PremiseNumber = xtra.DEWA_PREMISE;
            propInfo.isVacant = xtra.IS_RENTED == 0;
            propInfo.isLand = xtra.PROPERTY_TYPE_ID == 1;
        }
    });

    var prInfo = GetPropInfo(propId).then(function (prop) {
        if (prop && prop.Result) {
            propInfo.title = { en: prop.Result.PropertyEnglishDescription, ar: prop.Result.PropertyArabicDescription };
            propInfo.landStatus = landStatus[prop.Result.LandStatus];
            propInfo.landType = { en: prop.Result.LandTypeEnglish, ar: prop.Result.LandTypeArabic };
            propInfo.project = { en: prop.Result.ProjectEnglish, ar: prop.Result.ProjectArabic };
            propInfo.developer = { en: prop.Result.DeveloperEnglish, ar: prop.Result.DeveloperArabic };
            propInfo.rooms = { en: prop.Result.RoomsEnglish, ar: prop.Result.RoomsArabic };
            propInfo.propertySubType = { en: prop.Result.PropertySubTypeEnglish, ar: prop.Result.PropertySubTypeArabic };
            propInfo.landNo = prop.Result.LandNo;
            propInfo.landSubNo = prop.Result.LandSubNo;
            propInfo.dmNo = prop.Result.DMNo;
            propInfo.dmSubNo = prop.Result.DMSubNo;
            propInfo.canConvertToFH = prop.Result.CanConvertToFreehold;
        }
    });

    var prSvcChg = new Promise(function (resolve, reject) {
        getSreviceChargeIndexResultByTabuPropertyId(propId, function (serviceCharge) { resolve(serviceCharge); }, function () { resolve(null) })
    }).then(function (serviceCharge) {
        if (serviceCharge && serviceCharge.response && serviceCharge.response.InvoiceSummary) {
            propInfo.serviceChargeInfo = {
                serviceChargeDueAmount: serviceCharge.response.paymentDueAmount,
                serviceChargeInvoiceDate: serviceCharge.response.InvoiceSummary.invoiceDate
            };

        }
    });

    var prStatusInfo = GetPropStatusInfo(propId).then(function (info) {
        if (info && info.Result) {
            propInfo.ownershipStatus = info.Result.PropertyStatuses && info.Result.PropertyStatuses.length ? { en: info.Result.PropertyStatuses.map(i => i.EnglishStatus).join("-"), ar: info.Result.PropertyStatuses.map(i => i.ArabicStatus).join("-") } : null;
            propInfo.area = info.Result.Area;
            propInfo.lastTitledeed = info.Result.LastTitleDeed;
            propInfo.ownersCount = info.Result.NumberOfOwners;
            propInfo.isRestrained = info.Result.IsRestrained;
            propInfo.restrainList = info.Result.RestrainList;
            propInfo.isRegistered = info.Result.IsRegistered;
            propInfo.isMortgaged = info.Result.IsMortgaged;
            if (info.Result.IsMortgaged && info.Result.PropertyStatuses.length && info.Result.PropertyStatuses[0]) {
                propInfo.mortgageeName = info.Result.PropertyStatuses[0].MortgageeName;
            }
            propInfo.preRegistrationNumber = info.Result.PreRegistrationNumber;
            propInfo.isFreeHold = info.Result.IsFreeHold;
            propInfo.isBlocked = info.Result.BlockingHistory && info.Result.BlockingHistory.length > 0;
            propInfo.blockingHistory = (info.Result.BlockingHistory || []).map(function (i) {
                return {
                    Date: i.Date,
                    Authority: { ar: i.AuthorityAr, en: i.AuthorityEn },
                    Reason: i.Reason
                };
            });
            propInfo.blockingHistoryUnMapped = info.Result.BlockingHistory;
        }
    });

    var prRentalIdx = Promise.all([prXtra, prStatusInfo]).then(function () {
        return GetRentalIndex(propId, propInfo.isFreeHold, propInfo.propTypeId, propInfo.propSubTypeId).then(function (index) {
            if (index) {
                propInfo.rentalIndex = index;
            }
        });
    })

    var prPrjInfo = GetProjectInfo(propId).then(function (rsp) {
        if (rsp) {
            propInfo.projectInfo = rsp;
            propInfo.projectInfo.PROJECT_NAME = { en: propInfo.projectInfo.PROJECT_EN, ar: propInfo.projectInfo.PROJECT_AR }
            propInfo.projectInfo.DEVELOPER_NAME = { en: propInfo.projectInfo.DEVELOPER_EN, ar: propInfo.projectInfo.DEVELOPER_AR }
        }
    });

    var prMgmtInfo = GetMgmtInfo(propId).then(function (rsp) {
        if (rsp) {
            propInfo.mgmtInfo = {
                en: rsp.COMPANY_NAME_EN,
                ar: rsp.COMPANY_NAME_AR,
                phone: rsp.PHONE,
                email: rsp.EMAIL,
                address: rsp.ADDRESS
            }
        }
    });

    var prJopInfo = GetJOPMgmtInfo(propId).then(function (rsp) {
        if (rsp) {
            propInfo.jopMgmtInfo = {
                en: rsp.company.nameEn,
                ar: rsp.company.nameAr,
                phone: rsp.phone,
                email: rsp.email,
                address: rsp.address
            }
        }
    });

    var prZone = Promise.all([prXtra]).then(function () {
        if (propInfo.parcelId) {
            return GetZoneInfo(propInfo.parcelId).then(function (rsp) {
                if (rsp) {

                    propInfo.zoneInfo = {
                        height: { en: rsp.HEIGHT_REG_EN, ar: rsp.HEIGHT_REG_AR },
                        landUse: { en: rsp.LANDUSE_REG_EN, ar: rsp.LANDUSE_REG_AR },
                        parking: { en: rsp.PARKING_REG_EN, ar: rsp.PARKING_REG_AR },
                        setBack: { en: rsp.SETBACK_REG_EN, ar: rsp.SETBACK_REG_AR },
                        zoneCode: rsp.ZONE_CODE
                    };

                }
            });
        }
    });

    var prDmV = GetDMViolations(propId).then(function (violation) {
        if (violation && violation.length)
            propInfo.violation = violation.map(function (i) {
                return {
                    date: i.ViolationDate,
                    status: { en: i.ViolationStatusEng, ar: i.ViolationStatusArb },
                    description: { en: i.ViolationTypeEng, ar: i.ViolationTypeArb }
                };
            });
    });

    var prDewa = Promise.all([prXtra]).then(function () {
        if (propInfo.PremiseNumber) {
            return GetDEWADetails(propInfo.PremiseNumber).then(function (data) {
                if (data) {
                    propInfo.dewa = data;
                }
            })
        }
    });

    var prRdc = GetPropCaseInfo(propId).then(function (cases) {
        propInfo.cases = cases
    })

    //var prNFHInfo = GetNFH_FHInfo(propId).then(function (flag) {
    //    propInfo.canConvertToFH = flag
    //})


    Promise.all([prMap, prXtra, prInfo, prSvcChg, prStatusInfo, prPrjInfo, prMgmtInfo, prJopInfo, prRentalIdx, prDmV, prZone, prDewa, prRdc]).then(function () {
        const htmlTmpl = $('#tmp-property-details-result').html();
        const resultTemplate = Handlebars.compile(htmlTmpl);
        resultContainer.html(resultTemplate(propInfo));
        resultContainer.show();
        searchContainer.hide();
        $("#tx-loading").hide();

        if (propInfo.projectInfo) resultContainer.find("#btnPrjInfo, #btnDevInfo").removeAttr("disabled");
        if (propInfo.jopMgmtInfo) resultContainer.find("#btnJopMgmtComp").removeAttr("disabled");
        if (propInfo.mgmtInfo) resultContainer.find("#btnLeaseMgmtComp").removeAttr("disabled");
        if (propInfo.zoneInfo) resultContainer.find("#btnZoneInfo").removeAttr("disabled");

        tabify2('#projectStatusResultTabs');

        initMap(propInfo.lat, propInfo.lng);
        askForHappinessTransactional(4138);
        resetCaptcha();
    }).catch(function (err) {
        if (err == 'Invalid Captcha') {
            invalidCaptcha()
        } else {
            noDataFound()
        }
        resetCaptcha();
    });

}

function PropStatusSearchByLand(lang) {
    var subLand = $('#subLandNumber').val();
    if (subLand == '') {
        subLand = 0;
    }
    $("#tx-loading").show();

    if (AppInfo.featureGate.captchaEnabled && (!captcha1 || captcha1 == '')) {
        $("#captchaValidation1").html(tmplInvalidCaptcha({})).show();
        $("#tx-loading").hide();
        return;
    } else {
        $("#captchaValidation1").html(tmplInvalidCaptcha({})).hide();
    }

    var areaVal = $('#LandArea option:selected').val();
    var landReqUrl = landServiceUrl + "/GetByLandNumber?areaId=" + areaVal + "&landNumber=" + $('#landNumber').val() + "&landSubNumber=" + subLand;
    var buildingReqUrl = landServiceUrl + "/GetByBuildingNumber?areaId=" + areaVal + "&landNumber=" + $('#landNumber').val() + "&buildingNumber=" + $('#building-number').val();
    var unitReqUrl = landServiceUrl + "/GetByUnitNumber?buildingId=" + $('#building-lst option:selected').val() + "&unitNumber=" + $('#unit-number').val();

    var reqUrl = '';

    var propType = $('#property-type').val();

    if (propType === "1") {
        reqUrl = landReqUrl;
    } else if (propType === "2") {
        reqUrl = buildingReqUrl;
    } else {
        reqUrl = unitReqUrl;
    }

    if (AppInfo.featureGate.captchaEnabled && (!captcha1 || captcha1 == '')) {
        $('.land-result-wrap').html(invalidCaptcha());
        $('.land-result-wrap').show();
        $("#tx-loading").hide();
        return;
    }

    $.get(reqUrl + "&captchaResponse=" + captcha1 + "&serviceType=PropStatusSearchByLand", function (data) {
        if (data == "Invalid Captcha") {
            $("#captchaValidation").html(tmplInvalidCaptcha({})).show();
            $("#tx-loading").hide();
            return;
        } else {
            $("#captchaValidation").html(tmplInvalidCaptcha({})).hide();
        }
        var rsp = JSON.parse(data);
        if (rsp.Result) {
            getPropDetails(rsp.Result);
        }
        else {
            noDataFound();
        }
    });
}

var sortByProperty = function (property) {
    return function (x, y) {
        return ((x[property] === y[property]) ? 0 : ((x[property] > y[property]) ? 1 : -1));
    };
};

function filterAreas(query, lst, elemId, tempId) {
    var areas = lst;
    if (query && query.length > 0) {
        var q = query.toLowerCase();
        areas = areas.filter(function (a) {
            var name = a.name.toLowerCase();
            return name.indexOf(q) > -1;
        });
    }
    $(elemId).empty();
    var htmltmpl = $(tempId).html();
    var areaTemplate = Handlebars.compile(htmltmpl);
    areas.forEach(function (item) {
        $(elemId).append(areaTemplate({ val: item.val, name: item.name }));
    });
}

function InitGeoMap() {

    var uluru = { lng: 55.188713, lat: 25.050934 };
    geoMap = new google.maps.Map(document.getElementById('geo-map-search'), { zoom: 10, center: uluru, fullscreenControl: false });


    google.maps.event.addListener(geoMap, 'click', function (event) {
        $('#select_txt').hide();
        if (captcha4 && captcha4 != '') {
            GetProperties(event.latLng.lat(), event.latLng.lng());
        } else {
            mapLat = event.latLng.lat();
            mapLng = event.latLng.lng();
            captcha4 = 'retry';
            grecaptcha.reset(widget4);
            $('#google_captcha4').show();
        }
    });

    google.maps.event.addListener(geoMap, 'zoom_changed', function () {
        if ((captcha4 && captcha4 != '') || captchaHash != '') {
            var zoomLevel = geoMap.getZoom();
            if (zoomLevel >= 17 && !isZoomed) {
                var mylat = geoMap.getCenter().lat();
                var mylng = geoMap.getCenter().lng();
                GetProperties(mylat, mylng);

            } else if (zoomLevel < 17) {
                isZoomed = false;
            }
            if (zoomLevel < 16) {
                clearMarkers();
            }
        } else {
            mapLat = geoMap.getCenter().lat();
            mapLng = geoMap.getCenter().lng();
            captcha4 = 'retry';
            grecaptcha.reset(widget4);
            $('#google_captcha4').show();
        }
    });
}

function GetProperties(lat, lng) {
    $("#tx-loading").show();
    clearMarkers();
    $.get(
        landServiceUrl + '/GetLandByGeo',
        {
            'y_point': lat,
            'x_point': lng,
            'distance': 0.8,
            'captchaResponse': captcha4,
            'serviceType': 'SearchOnMap'
        },
        function (data) {
            if (data != '') {
                var jsonObj = JSON.parse(data);
                jsonObj = JSON.parse(jsonObj);
                if (jsonObj.length > 0) {

                    locations = [];
                    clearMarkers();
                    var promises = jsonObj.map(function (ele) {
                        //var result = await GetPropInfo(ele.PropId).then(function (prop) {
                        locations.push({ info: '', lat: ele.Point_Y, lng: ele.Point_X, propId: ele.PropId, landNumber: ele.LandNumber, mun: ele.Mun, makani: ele.Makani });
                        //});
                        // return new Promise(function (res, rej) { res(result) });
                    });

                    Promise.all(promises)
                        .then(function (results) {
                            drawLocations();
                            isZoomed = true;
                        })
                } else {
                    bindNoDataTmp();
                    $("#tx-loading").hide();
                }
            } else {
                bindNoDataTmp();
                $("#tx-loading").hide();
            }
            resetCaptcha(true);
        }
    );
}
function drawLocations() {
    var infowindow = new google.maps.InfoWindow();

    var marker, i;
    var image = {
        url: 'http://maps.google.com/mapfiles/ms/icons/red-dot.png',
        // This marker is 20 pixels wide by 32 pixels tall.
        size: new google.maps.Size(100, 30),
        // The origin for this image is 0,0.
        origin: new google.maps.Point(0, 0)
    };

    for (i = 0; i < locations.length; i++) {
        marker = new google.maps.Marker({
            position: new google.maps.LatLng(locations[i].lat, locations[i].lng),
            map: geoMap,
            icon: image,
            title: locations[i].mun
        });
        markers.push(marker);
        google.maps.event.addListener(marker, 'click', (function (marker, i) {
            return function () {
                showLocationResult(locations[i].propId, locations[i].landNumber, locations[i].mun, locations[i].makani);
            }
        })(marker, i));
    }
    $("#tx-loading").hide();
}

function clearMarkers() {
    for (var i = 0; i < markers.length; i++) {
        markers[i].setMap(null);
    }
    markers = [];
}

function exportReport() {
    var exportData = {};
    exportData.ownershipStatus = propInfo.ownershipStatus.ar;
    exportData.area = propInfo.area;
    exportData.lastTitledeed = propInfo.lastTitledeed;
    exportData.ownersCount = propInfo.ownersCount;
    exportData.mortgageeName = propInfo.mortgageeName;
    exportData.isRestrained = propInfo.isRestrained;
    exportData.mortgage = propInfo.isMortgaged;
    exportData.status = propInfo.title.ar;
    exportData.preRegistrationNumber = propInfo.preRegistrationNumber;
    exportData.isFreeHold = propInfo.isFreeHold;
    exportData.isRegistered = propInfo.isRegistered;
    exportData.isBlocked = propInfo.isBlocked;
    exportData.blockingHistory = propInfo.blockingHistoryUnMapped;
    exportData.landStatus = propInfo.landStatus.ArabicName;
    exportData.landType = propInfo.landType.ar;
    exportData.project = propInfo?.projectInfo?.PROJECT_AR;
    exportData.developer = propInfo?.projectInfo?.DEVELOPER_AR;
    exportData.rooms = propInfo.rooms.ar;
    exportData.propertySubType = propInfo.propertySubType.ar;
    exportData.landNo = propInfo.landNo;
    exportData.landSubNo = propInfo.landSubNo;
    exportData.dmNo = propInfo.dmNo;
    exportData.dmSubNo = propInfo.dmSubNo;
    exportData.IsFreeHoldCovertableValue = propInfo.canConvertToFH;

    if (propInfo.serviceChargeInfo) {
        exportData.isServiceChargeAvailable = true;
        exportData.serviceChargeDueAmount = formatNumber(propInfo.serviceChargeInfo.serviceChargeDueAmount);
        exportData.serviceChargeInvoiceDate = moment(propInfo.serviceChargeInfo.serviceChargeInvoiceDate).format("DD/MM/YYYY");
    } else {
        exportData.isServiceChargeAvailable = false;
    }

    if (propInfo.isRestrained) {
        if (propInfo.restrainList) {
            exportData.restrainList = [...propInfo.restrainList];
            exportData.restrainList.forEach(function (data, index) {
                data.Serail = ++index;
                data.CaseRegistrationDate = moment(data.CaseRegistrationDate).format("DD/MM/YYYY");
            });
        }
    }

    exportPdf(exportData);
}
